// Agente B · duplo toque no nível da API: duas requisições IGUAIS ao mesmo tempo, cada uma com a sua
// Idempotency-Key (é o que acontece se o botão não travar: a chave nasce por clique). Mede o que o
// SERVIDOR faz sozinho, sem a trava do botão. Banco LOCAL, tenant do agente B.
import { randomUUID } from 'node:crypto';
import { cookieHeader, psql } from './B-lib-log.mjs';
const BASE = 'http://localhost:3200';
const cookie = cookieHeader();
const T = '320d83de-611f-48e1-9761-570863fbb10b';
const post = (path, body, anon = false) => fetch(BASE + path, { method: 'POST', headers: { ...(anon ? {} : { cookie }), 'content-type': 'application/json', origin: BASE, 'idempotency-key': randomUUID() }, body: JSON.stringify(body ?? {}) }).then(async (r) => [r.status, (await r.text()).slice(0, 120)]);
const n = (sql) => Number(psql(sql));
// 1) Chamar cliente (recover/manual): o botão da tela não trava (fire-and-forget)
const cli = '3a27a6d8-c442-4674-a9b4-2e952342a228';
let a = n(`select count(*) from messages where tenant_id='${T}' and client_id='${cli}'`);
let r = await Promise.all([post('/api/v1/cycle/recover/manual', { clientId: cli }), post('/api/v1/cycle/recover/manual', { clientId: cli })]);
console.log('chamar cliente ×2 simultâneo:', r.map((x) => x[0]).join(','), '→ mensagens gravadas:', n(`select count(*) from messages where tenant_id='${T}' and client_id='${cli}'`) - a);
// 2) Criar agendamento ×2 (mesmo horário): a constraint de sobreposição deve segurar
const corpo = { clientId: cli, serviceId: '083649fa-03b0-4ed3-86c1-77f39b978aee', professionalId: 'd5047e62-6b51-434c-b4a2-457904a69e15', startsAt: `2026-10-20T${10 + Math.floor(Math.random() * 8)}:${Math.random() < 0.5 ? '00' : '30'}:00-03:00` };
a = n(`select count(*) from appointments where tenant_id='${T}'`);
r = await Promise.all([post('/api/v1/appointments', corpo), post('/api/v1/appointments', corpo)]);
console.log('criar agendamento ×2 simultâneo:', r.map((x) => x[0]).join(','), '→ criados:', n(`select count(*) from appointments where tenant_id='${T}'`) - a, '| 2ª resposta:', r.find((x) => x[0] !== 200)?.[1] ?? '-');
// 3) Adicionar item ×2 na mesma comanda (sem trava do botão)
const ap = (await (await fetch(BASE + '/api/v1/appointments', { method: 'POST', headers: { cookie, 'content-type': 'application/json', origin: BASE, 'idempotency-key': randomUUID() }, body: JSON.stringify({ ...corpo, startsAt: corpo.startsAt.replace('2026-10-20', '2026-10-21') }) })).json()).data.appointment.id;
for (const p of ['confirm', 'arrive']) await post(`/api/v1/appointments/${ap}/${p}`);
const tk = JSON.parse((await post(`/api/v1/appointments/${ap}/complete`))[1].length ? '{}' : '{}');
void tk;
const ticket = psql(`select id from tickets where appointment_id='${ap}'`);
a = n(`select count(*) from ticket_items where ticket_id='${ticket}'`);
r = await Promise.all([post(`/api/v1/tickets/${ticket}/items`, { serviceId: corpo.serviceId, professionalId: corpo.professionalId, qty: 1 }), post(`/api/v1/tickets/${ticket}/items`, { serviceId: corpo.serviceId, professionalId: corpo.professionalId, qty: 1 })]);
console.log('adicionar item ×2 simultâneo:', r.map((x) => x[0]).join(','), '→ itens gravados:', n(`select count(*) from ticket_items where ticket_id='${ticket}'`) - a);
// 4) Fechar comanda ×2
a = n(`select count(*) from audit_log where entity_id='${ticket}' and action='ticket.close'`);
r = await Promise.all([post(`/api/v1/tickets/${ticket}/close`, { paymentMethod: 'pix' }), post(`/api/v1/tickets/${ticket}/close`, { paymentMethod: 'pix' })]);
console.log('fechar comanda ×2 simultâneo:', r.map((x) => x[0]).join(','), '→ fechamentos auditados:', n(`select count(*) from audit_log where entity_id='${ticket}' and action='ticket.close'`) - a, '| monthly_profit linhas do mês:', psql(`select count(*) from monthly_profit where tenant_id='${T}'`));
