import { chromium } from "playwright";
// APR-021/022: enfileirar sem rede → fica na tela com aviso; fechar, abrir com rede → POST em < 3 s e fila vazia.
const B = "http://localhost:3201";
const SENHA = process.env.SEED_SENHA ?? "DemoCiclo#2026";
const CAPT = "C:/Users/Usuario/.claude/code/ciclo-aprimoramento/docs/98-aprimoramento/capturas/";
const CELULAR = { viewport: { width: 390, height: 844 }, hasTouch: true, isMobile: true, timezoneId: "America/Sao_Paulo", locale: "pt-BR" };
const contar = () =>
  new Promise((ok) => {
    const req = indexedDB.open("ciclo-offline");
    req.onsuccess = () => {
      try {
        const c = req.result.transaction("mutations", "readonly").objectStore("mutations").count();
        c.onsuccess = () => ok(c.result);
      } catch (e) {
        ok("erro " + e);
      }
    };
  });
const dia = new Date(Date.now() + 3 * 86_400_000).toISOString().slice(0, 10);
const b = await chromium.launch({ channel: "chrome" });
const rodadas = [];
for (let i = 0; i < 5; i++) {
  const ctx = await b.newContext(CELULAR);
  const p = await ctx.newPage();
  await p.goto(B + "/entrar");
  await p.fill('input[type="email"]', "dono-demo-dom-estilo@ciclo.app");
  await p.fill('input[type="password"]', SENHA);
  await Promise.all([p.waitForURL(/admin/, { timeout: 60000 }), p.press('input[type="password"]', "Enter")]);
  await p.goto(B + "/admin/agenda/novo", { waitUntil: "networkidle" });
  await ctx.setOffline(true);
  await p.locator('input[placeholder="Nome"]').fill(`Fila Offline ${i + 1}`);
  await p.locator('input[type=tel], input[placeholder*="98765"]').first().fill(`(11) 90000-04${String(50 + i)}`);
  await p.locator('input[type="datetime-local"]').first().fill(`${dia}T${String(9 + i * 2).padStart(2, "0")}:15`);
  const navegacoes = [];
  p.on("framenavigated", (f) => { if (f === p.mainFrame()) navegacoes.push(f.url()); });
  await p.getByRole("button", { name: /Confirmar agendamento|Marcar horário/ }).click();
  await p.waitForTimeout(2500);
  const aviso = (await p.locator("main").innerText()).match(/Guardado neste aparelho[^\n]*/)?.[0] ?? null;
  const urlDepois = new URL(p.url()).pathname;
  const naFila = await p.evaluate(contar);
  if (i === 0) await p.screenshot({ path: CAPT + "apr022-guardado-offline-390.png" });
  await p.close();
  // Volta a rede e ABRE o app de novo: a fila sai sozinha?
  await ctx.setOffline(false);
  const p2 = await ctx.newPage();
  const posts = [];
  const t0 = Date.now();
  p2.on("request", (q) => { if (q.method() === "POST" && q.url().includes("/api/v1/appointments")) posts.push(Date.now() - t0); });
  await p2.goto(B + "/admin/hoje", { waitUntil: "domcontentloaded" });
  await p2.waitForTimeout(5000);
  const aindaNaFila = await p2.evaluate(contar);
  rodadas.push({ rodada: i + 1, urlDepois, aviso, navegacoesErro: navegacoes.filter((u) => u.startsWith("chrome-error")).length, naFila, postEmMs: posts[0] ?? null, aindaNaFila });
  await ctx.close();
}
console.log(JSON.stringify(rodadas, null, 1));
await b.close();
