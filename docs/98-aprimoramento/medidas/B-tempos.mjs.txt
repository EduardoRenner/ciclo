function chaveLocal() {
  const env = readFileSync('C:/Users/Usuario/.claude/code/ciclo-aprimoramento/.env.local', 'utf8');
  const get = (k) => new RegExp(`^${k}=(.*)$`, 'm').exec(env)?.[1]?.trim().replace(/^"|"$/g, '');
  return get('PUBLIC_LINK_SIGNING_KEY') || get('CRON_SECRET');
}
function tokenConfirmacao(id) {
  const exp = Date.now() + 72 * 3_600_000;
  const payload = `confirmacao_agendamento.${id}.${exp}`;
  const ass = createHmac('sha256', chaveLocal()).update(payload, 'utf8').digest('hex');
  return Buffer.from(`${payload}.${ass}`, 'utf8').toString('base64url');
}

// Predicado de "conteúdo útil" (a CSP do app proíbe eval, então o predicado é uma função fixa + spec).
// tipo: 'h2' (algum h2 casa), 'texto' (innerText do <main> casa), 'sel' (seletor existe), 'botao' (algum botão casa).
const H2 = (re) => ({ tipo: 'h2', re });
const TEXTO = (re) => ({ tipo: 'texto', re });
const SEL = (sel) => ({ tipo: 'sel', sel });
const BOTAO = (re) => ({ tipo: 'botao', re });
function predicado(spec) {
  const re = spec.re ? new RegExp(spec.re, 'i') : null;
  let ok = false;
  if (spec.tipo === 'h2') ok = [...document.querySelectorAll('h2')].some((h) => re.test(h.textContent || ''));
  else if (spec.tipo === 'texto') ok = re.test((document.querySelector('main') || document.body).innerText || '');
  else if (spec.tipo === 'sel') ok = !!document.querySelector(spec.sel);
  else if (spec.tipo === 'botao') ok = [...document.querySelectorAll('button')].some((b) => re.test(b.textContent || ''));
  return ok ? performance.now() : false;
}

const INIT = () => {
  window.__m = { cls: 0, longas: [], lcp: 0 };
  try {
    new PerformanceObserver((l) => {
      for (const e of l.getEntries()) if (!e.hadRecentInput) window.__m.cls += e.value;
    }).observe({ type: 'layout-shift', buffered: true });
  } catch {}
  try {
    new PerformanceObserver((l) => {
      for (const e of l.getEntries()) window.__m.longas.push(Math.round(e.duration));
    }).observe({ type: 'longtask', buffered: true });
  } catch {}
  try {
    new PerformanceObserver((l) => {
      for (const e of l.getEntries()) window.__m.lcp = e.startTime;
    }).observe({ type: 'largest-contentful-paint', buffered: true });
  } catch {}
};

async function prepararPagina(ctx, perfil) {
  const page = await ctx.newPage();
  const cdp = await ctx.newCDPSession(page);
  if (perfil.cpu > 1) await cdp.send('Emulation.setCPUThrottlingRate', { rate: perfil.cpu });
  if (perfil.rede) {
    await cdp.send('Network.enable');
    await cdp.send('Network.emulateNetworkConditions', { offline: false, ...perfil.rede });
  }
  return page;
}

async function medirUma(page, url, util) {
  const t0 = Date.now();
  await page.goto(url, { waitUntil: 'commit', timeout: 120000 });
  let utilMs = null;
  try {
    // Sonda pelo CDP (evaluate de função não passa por eval, então a CSP do app não bloqueia).
    const fim = Date.now() + 60000;
    while (Date.now() < fim) {
      const v = await page.evaluate(predicado, util).catch(() => false);
      if (v) {
        utilMs = Math.round(v);
        break;
      }
      await new Promise((r) => setTimeout(r, 25));
    }
  } catch (e) {
    if (process.env.DEBUG) console.log('util falhou:', String(e).slice(0, 200));
  }
  await page.waitForLoadState('networkidle', { timeout: 60000 }).catch(() => {});
  await page.waitForTimeout(300);
  const r = await page.evaluate(() => {
    const nav = performance.getEntriesByType('navigation')[0];
    const fcp = performance.getEntriesByName('first-contentful-paint')[0]?.startTime ?? null;
    const res = performance.getEntriesByType('resource');
    const js = res.filter((e) => e.initiatorType === 'script' || /\.js(\?|$)/.test(e.name));
    return {
      url: location.pathname,
      ttfb: nav ? Math.round(nav.responseStart) : null,
      docTransfer: nav ? nav.transferSize : null,
      fcp: fcp && Math.round(fcp),
      lcp: Math.round(window.__m?.lcp || 0),
      cls: Math.round((window.__m?.cls || 0) * 1000) / 1000,
      longas: window.__m?.longas || [],
      jsTransferB: js.reduce((s, e) => s + (e.transferSize || 0), 0),
      jsDecodedB: js.reduce((s, e) => s + (e.decodedBodySize || 0), 0),
      jsArquivos: js.length,
      requisicoes: res.length + 1,
      carga: nav ? Math.round(nav.loadEventEnd) : null,
    };
  });
  return { ...r, utilMs, paredeMs: Date.now() - t0 };
}

const mediana = (v) => {
  const a = v.filter((x) => x != null).sort((x, y) => x - y);
  if (!a.length) return null;
  const m = Math.floor(a.length / 2);
  return a.length % 2 ? a[m] : Math.round((a[m - 1] + a[m]) / 2);
};
