// ---------- preparação de dados pela API (não medida) ----------
async function api(method, path, body) {
  const r = await fetch(BASE + path, { method, headers: { cookie, 'content-type': 'application/json', origin: BASE, 'idempotency-key': randomUUID() }, body: body ? JSON.stringify(body) : undefined });
  const t = await r.text();
  try {
    return { status: r.status, json: JSON.parse(t) };
  } catch {
    return { status: r.status, json: null };
  }
}
let slot = 0;
async function criarPendente(dia = '2026-10-16') {
  for (let tent = 0; tent < 40; tent++) {
    const i = slot++;
    const prof = PROFS[i % 3];
    const hora = 9 + (Math.floor(i / 3) % 10);
    const r = await api('POST', '/api/v1/appointments', { clientId: CLIENTES[i % 3], serviceId: SERVICO, professionalId: prof, startsAt: `${dia}T${String(hora).padStart(2, '0')}:00:00-03:00` });
    if (r.status === 200) return r.json.data.appointment.id;
  }
  throw new Error('não achei horário livre para preparar');
}
async function abrirComanda() {
  const id = await criarPendente('2026-10-13');
  await api('POST', `/api/v1/appointments/${id}/confirm`);
  await api('POST', `/api/v1/appointments/${id}/arrive`);
  const c = await api('POST', `/api/v1/appointments/${id}/complete`);
  return c.json.data.ticket.id;
}

// ---------- instrumentação na página ----------
function instalar() {
  if (window.__i) return;
  window.__i = { armado: false, mut: null, ev: [], alvoMudou: null };
  try {
    new PerformanceObserver((l) => {
      for (const e of l.getEntries()) if (window.__i.armado) window.__i.ev.push({ n: e.name, s: e.startTime, d: e.duration, id: e.interactionId });
    }).observe({ type: 'event', durationThreshold: 16, buffered: false });
  } catch {}
  new MutationObserver(() => {
    const i = window.__i;
    if (!i.armado) return;
    const t = performance.now();
    if (i.mut == null) i.mut = t;
    const alvo = document.querySelector('[data-b-alvo]');
    if (i.alvoMudou == null && (!alvo || alvo.disabled || alvo.getAttribute('aria-busy') === 'true' || alvo.getAttribute('aria-disabled') === 'true')) i.alvoMudou = t;
  }).observe(document.documentElement, { subtree: true, childList: true, attributes: true, characterData: true });
}
function armar() {
  const i = window.__i;
  i.mut = null;
  i.ev = [];
  i.alvoMudou = null;
  i.t0 = performance.now();
  i.armado = true;
}
function ler() {
  const i = window.__i;
  i.armado = false;
  return { t0: i.t0, mut: i.mut, ev: i.ev, alvoMudou: i.alvoMudou };
}
function predicado(spec) {
  const re = spec.re ? new RegExp(spec.re, 'i') : null;
  let ok = false;
  if (spec.tipo === 'texto') ok = re.test((document.querySelector('main') || document.body).innerText || '');
  else if (spec.tipo === 'body') ok = re.test(document.body.innerText || '');
  else if (spec.tipo === 'sel') ok = !!document.querySelector(spec.sel);
  else if (spec.tipo === 'url') ok = re.test(location.pathname + location.search);
  return ok ? performance.now() : false;
}
async function esperar(page, spec, ms = 60000) {
  const fim = Date.now() + ms;
  while (Date.now() < fim) {
    const v = await page.evaluate(predicado, spec).catch(() => false);
    if (v) return v;
    await new Promise((r) => setTimeout(r, 20));
  }
  return null;
}

async function novaPagina(ctx) {
  const page = await ctx.newPage();
  const cdp = await ctx.newCDPSession(page);
  if (perfil.cpu > 1) await cdp.send('Emulation.setCPUThrottlingRate', { rate: perfil.cpu });
  if (perfil.rede) {
    await cdp.send('Network.enable');
    await cdp.send('Network.emulateNetworkConditions', { offline: false, ...perfil.rede });
  }
  page.__posts = [];
  page.on('request', (r) => {
    if (r.method() !== 'GET' && r.url().includes('/api/')) page.__posts.push({ url: new URL(r.url()).pathname.replace(/[0-9a-f-]{36}/g, ':id'), t: Date.now() });
  });
  return page;
}

async function tocar(page, loc) {
  if (perfil.hasTouch) await loc.tap();
  else await loc.click();
}
async function tocarDuasVezes(page, loc) {
  const box = await loc.boundingBox();
  const x = box.x + box.width / 2;
  const y = box.y + box.height / 2;
  // Entrada crua, sem esperar o botão ficar habilitado: é o que o dedo faz.
  if (perfil.hasTouch) {
    await page.touchscreen.tap(x, y);
    await page.touchscreen.tap(x, y);
  } else {
    await page.mouse.click(x, y);
    await page.mouse.click(x, y);
  }
}

/** Mede uma interação: prepara(page) → localizador do alvo → toque → espera estado final. */
async function medir(page, { nome, alvo, final, duplo = false }) {
  const loc = typeof alvo === 'function' ? alvo(page) : page.locator(alvo).first();
  await loc.waitFor({ state: 'visible', timeout: 60000 });
  await loc.evaluate((el) => el.setAttribute('data-b-alvo', '1'));
  await page.evaluate(instalar);
  await page.waitForTimeout(150);
  await page.evaluate(armar);
  const postsAntes = page.__posts.length;
  if (duplo) await tocarDuasVezes(page, loc);
  else await tocar(page, loc);
  const tFinal = await esperar(page, final);
  await page.waitForTimeout(duplo ? 3000 : 400);
  const r = await page.evaluate(ler).catch(() => null);
  if (!r) return { nome, erro: 'página trocou antes da leitura (navegação)', posts: page.__posts.slice(postsAntes).map((p) => p.url) };
  const inicio = r.ev.length ? Math.min(...r.ev.map((e) => e.s)) : r.t0;
  const inp = r.ev.length ? Math.round(Math.max(...r.ev.map((e) => e.d))) : '<16';
  return {
    nome,
    // `inicio` = 1º evento que o Event Timing registrou (só registra eventos > 16 ms; se o pointerdown foi
    // rápido, o 1º registrado pode ser o click, e a mudança aparece "antes" dele: negativo = imediato).
    primeiraMudancaMs: r.mut != null ? Math.round(r.mut - inicio) : null,
    primeiraMudancaDesdeArmarMs: r.mut != null ? Math.round(r.mut - r.t0) : null,
    alvoReagiuMs: r.alvoMudou != null ? Math.round(r.alvoMudou - inicio) : null,
    inpMs: inp,
    finalMs: tFinal != null ? Math.round(tFinal - inicio) : null,
    posts: page.__posts.slice(postsAntes).map((p) => p.url),
    alertas: (await page.locator('[role=alert]').allInnerTexts().catch(() => [])).join(' | ').slice(0, 200) || undefined,
  };
}

const mediana = (v) => {
  const a = v.filter((x) => typeof x === 'number').sort((x, y) => x - y);
  if (!a.length) return null;
  const m = Math.floor(a.length / 2);
  return a.length % 2 ? a[m] : Math.round((a[m - 1] + a[m]) / 2);
};
const pior = (v) => {
  const a = v.filter((x) => typeof x === 'number');
  return a.length ? Math.max(...a) : null;
};

async function prepararNovo(page, k) {
  try {
    const sels = page.locator('main select');
    await sels.first().waitFor({ timeout: 60000 });
    await sels.nth(0).selectOption(SERVICO);
    await sels.nth(1).selectOption(PROFS[k % 3]).catch(() => {});
    const hora = String(9 + (k % 10)).padStart(2, '0');
    const quando = `2026-10-17T${hora}:${k % 2 ? '30' : '00'}`;
    await page.locator('input[type="datetime-local"]').fill(quando);
    // ?cliente= é pré-preenchido por um GET no cliente (useEffect); sem esperar, o formulário recusa.
    for (let t = 0; t < 600; t++) {
      const ok = await page.evaluate(() => [...document.querySelectorAll('main input[type=text]')].some((i) => i.value.length > 1)).catch(() => false);
      if (ok) break;
      await page.waitForTimeout(100);
    }
    await page.waitForTimeout(400);
    return quando;
  } catch (e) {
    console.log('prepararNovo falhou', String(e).slice(0, 200));
    return null;
  }
}
