import { chromium } from 'file:///C:/temp/ciclo-medicao/node_modules/playwright/index.mjs'
import fs from 'node:fs'

const ESTADO = 'C:/temp/ciclo-medicao/estado-A-dono.json' // dono demo-dom-estilo, fora do repositório
const CAPT = 'C:/Users/Usuario/.claude/code/ciclo-aprimoramento/docs/98-aprimoramento/capturas/'
const SAIDA = 'C:/Users/Usuario/.claude/code/ciclo-aprimoramento/docs/98-aprimoramento/medidas/fase2-reproducoes.json'
const CELULAR = { viewport: { width: 390, height: 844 }, hasTouch: true, isMobile: true, timezoneId: 'America/Sao_Paulo', locale: 'pt-BR' }

const horas = (t) => [...t.matchAll(/\b([01]\d|2[0-3]):[0-5]\d\b/g)].map((m) => m[0])

async function c07() {
  const r = {}
  for (const [rotulo, base] of [['servidor_sp_3200', 'http://localhost:3200'], ['servidor_utc_3201', 'http://localhost:3201']]) {
    const ctx = await navegador.newContext({ ...CELULAR, storageState: ESTADO, javaScriptEnabled: false })
    const p = await ctx.newPage()
    r[rotulo] = {}
    for (const rota of ['/admin/agenda?date=2026-10-14', '/admin/hoje']) {
      await p.goto(base + rota, { waitUntil: 'load' })
      const texto = await p.locator('main').innerText().catch(() => p.locator('body').innerText())
      r[rotulo][rota] = horas(texto)
    }
    await p.goto(base + '/admin/config/horarios', { waitUntil: 'load' })
    r[rotulo].folga_inicio_padrao_html = await p.locator('input[name=inicio]').first().getAttribute('value')
    await p.screenshot({ path: CAPT + `fase2-C07-agenda-sem-js-${rotulo}.png`, fullPage: false }).catch(() => {})
    await ctx.close()
  }
  // Com JS: o que a pessoa vê depois da hidratação, e o erro de hidratação no console.
  const ctx = await navegador.newContext({ ...CELULAR, storageState: ESTADO })
  const p = await ctx.newPage()
  const erros = []
  p.on('console', (m) => { if (m.type() === 'error') erros.push(m.text().slice(0, 160)) })
  p.on('pageerror', (e) => erros.push('pageerror: ' + String(e.message).slice(0, 160)))
  await p.goto('http://localhost:3201/admin/agenda?date=2026-10-14', { waitUntil: 'networkidle' })
  r.utc_com_js_depois_de_hidratar = horas(await p.locator('main').innerText())
  r.utc_com_js_erros_console = erros
  await ctx.close()
  r.relogio_maquina = new Date().toString()
  return r
}

async function c04() {
  const slug = 'demo-dom-estilo'
  const serviceId = 'e5ee748d-24b7-4bba-8387-8ffecc1277fb' // Corte
  const disp = await (await fetch(`http://localhost:3200/api/v1/public/${slug}/availability?serviceId=${serviceId}&date=2026-10-22`)).json()
  const slot = disp.data.slots.at(-1)
  const corpo = { serviceId, professionalId: slot.professionalId, startsAt: slot.startsAt, name: 'Teste Fase2 Reenvio', phone: '(11) 90000-0420' }
  const enviar = () => fetch(`http://localhost:3200/api/v1/public/${slug}/book`, { method: 'POST', headers: { 'content-type': 'application/json' }, body: JSON.stringify(corpo) })
  const r1 = await enviar()
  const j1 = await r1.json()
  // Simula: a resposta do 1º se perdeu na rede e a cliente toca "Confirmar" de novo com o MESMO corpo.
  const r2 = await enviar()
  const j2 = await r2.json()
  return { slot: slot.startsAt, primeiro: { status: r1.status, appointmentId: j1.data?.appointmentId ?? null }, reenvio: { status: r2.status, code: j2.error?.code, message: j2.error?.message } }
}

async function c01() {
  const ctx = await navegador.newContext({ ...CELULAR, storageState: ESTADO })
  const req = ctx.request
  const id = 'e32a7863-0780-433c-81bd-6756746753ae' // Corte + barba, R$ 70,00, 14/10 09:11, confirmado (seed)
  const h = () => ({ 'content-type': 'application/json', 'idempotency-key': crypto.randomUUID() })
  const chegou = await req.post(`http://localhost:3200/api/v1/appointments/${id}/arrive`, { headers: h(), data: {} })
  const concluiu = await req.post(`http://localhost:3200/api/v1/appointments/${id}/complete`, { headers: h(), data: {} })
  const jc = await concluiu.json().catch(() => ({}))
  const ticketId = jc.data?.ticket?.id ?? null
  const r = { chegou: chegou.status(), concluiu: concluiu.status(), ticketId }
  if (ticketId) {
    const p = await ctx.newPage()
    await p.goto(`http://localhost:3200/admin/comanda/${ticketId}`, { waitUntil: 'networkidle' })
    const texto = await p.locator('main').innerText()
    r.comanda_vazia = /Nenhum item ainda/.test(texto)
    r.servico_pre_selecionado = await p.locator('select').first().evaluate((s) => s.options[s.selectedIndex]?.text ?? null).catch(() => null)
    r.servico_do_agendamento = 'Corte + barba · R$ 70,00'
    await p.screenshot({ path: CAPT + 'fase2-C01-comanda-nasce-vazia-390.png' })
    await p.close()
  }
  await ctx.close()
  resultado._ticket_c01 = ticketId
  return r
}

async function c13() {
  const ticketId = resultado._ticket_c01
  if (!ticketId) return { pulado: 'sem comanda aberta da parte c01' }
  const ctx = await navegador.newContext({ ...CELULAR, storageState: ESTADO })
  const p = await ctx.newPage()
  await p.goto(`http://localhost:3200/admin/comanda/${ticketId}`, { waitUntil: 'networkidle' })
  // lança o serviço certo (Corte + barba) para poder fechar
  await p.locator('select').first().selectOption({ label: 'Corte + barba · R$ 70,00' }).catch(async () => {
    const opcoes = await p.locator('select').first().locator('option').allInnerTexts()
    await p.locator('select').first().selectOption({ label: opcoes.find((o) => o.startsWith('Corte + barba')) })
  })
  await p.getByRole('button', { name: 'Adicionar', exact: true }).click()
  await p.getByRole('button', { name: /Remover/ }).first().waitFor({ timeout: 8000 })
  await p.getByRole('button', { name: 'Pix', exact: true }).tap()
  const desconto = p.locator('label:has-text("Desconto") input')
  await desconto.tap()
  await desconto.fill('10')
  const posts = []
  p.on('request', (q) => { if (q.method() !== 'GET' && q.url().includes('/api/v1/tickets')) posts.push(q.method() + ' ' + q.url().replace(/.*\/api\/v1/, '')) })
  // UM toque em "Fechar comanda" com o campo de desconto ainda focado (o blur dispara o PATCH)
  await p.getByRole('button', { name: 'Fechar comanda' }).tap()
  await p.waitForTimeout(4000)
  const texto = await p.locator('main').innerText()
  const r = { requisicoes_depois_do_toque: posts, comanda_fechou_com_um_toque: /Comanda fechada/.test(texto) }
  await p.screenshot({ path: CAPT + 'fase2-C13-fechar-com-desconto-focado-390.png' })
  await ctx.close()
  return r
}

async function c45() {
  const ctx = await navegador.newContext({ ...CELULAR, storageState: ESTADO })
  const p = await ctx.newPage()
  await p.goto('http://localhost:3200/admin/agenda/novo', { waitUntil: 'networkidle' })
  await ctx.setOffline(true)
  await p.locator('input[placeholder="Nome"]').fill('Teste Fase2 Fila')
  await p.locator('input[type=tel], input[placeholder*="98765"]').first().fill('(11) 90000-0445')
  await p.locator('input[type="datetime-local"]').first().fill('2026-10-23T16:30')
  await p.getByRole('button', { name: /Confirmar agendamento|Marcar horário/ }).click()
  await p.waitForTimeout(2500)
  const depoisDeEnfileirar = (await p.locator('body').innerText()).slice(0, 300)
  const urlDepoisDeEnfileirar = p.url()
  await p.screenshot({ path: CAPT + 'fase2-C48-depois-de-enfileirar-offline-390.png' })
  const naFila = 'medido na pagina seguinte (a atual virou pagina de erro do navegador)'
  await p.close()
  // Volta a rede e ABRE o app de novo (sem evento `online` na página nova): a fila sai sozinha?
  await ctx.setOffline(false)
  const p2 = await ctx.newPage()
  const posts = []
  p2.on('request', (q) => { if (q.method() === 'POST' && q.url().includes('/api/v1/appointments')) posts.push(q.url()) })
  await p2.goto('http://localhost:3200/admin/hoje', { waitUntil: 'domcontentloaded' })
  const naFilaAoAbrir = await p2.evaluate(() => new Promise((ok) => { const req = indexedDB.open('ciclo-offline'); req.onsuccess = () => { try { const c = req.result.transaction('mutations', 'readonly').objectStore('mutations').count(); c.onsuccess = () => ok(c.result) } catch (e) { ok('erro ' + e) } } }))
  await p2.waitForTimeout(10000)
  const faixa = (await p2.locator('body').innerText()).match(/[^\n]*(pendente|enviando|guardad)[^\n]*/i)?.[0] ?? null
  const aindaNaFila = await p2.evaluate(() => new Promise((ok) => {
    const req = indexedDB.open('ciclo-offline')
    req.onsuccess = () => { try { const tx = req.result.transaction('mutations', 'readonly'); const c = tx.objectStore('mutations').count(); c.onsuccess = () => ok(c.result) } catch (e) { ok('erro ' + e) } }
  }))
  await p2.screenshot({ path: CAPT + 'fase2-C45-fila-nao-sai-ao-abrir-390.png' })
  await ctx.close() // o contexto descartável leva o IndexedDB junto: nada fica para reenviar depois
  return { url_depois_de_enfileirar: urlDepoisDeEnfileirar, depois_de_enfileirar: depoisDeEnfileirar, itens_na_fila_offline: naFila, itens_na_fila_ao_abrir_online: naFilaAoAbrir, posts_em_10s_depois_de_abrir_online: posts.length, itens_ainda_na_fila: aindaNaFila, faixa }
}
