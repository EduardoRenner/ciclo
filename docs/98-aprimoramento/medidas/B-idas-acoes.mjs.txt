// Lê a chave de assinatura do .env.local SEM imprimir (só para gerar o link de confirmação local).
function chaveLocal() {
  const env = readFileSync('C:/Users/Usuario/.claude/code/ciclo-aprimoramento/.env.local', 'utf8');
  const get = (k) => new RegExp(`^${k}=(.*)$`, 'm').exec(env)?.[1]?.trim().replace(/^"|"$/g, '');
  return get('PUBLIC_LINK_SIGNING_KEY') || get('CRON_SECRET');
}
function tokenConfirmacao(id) {
  const exp = Date.now() + 72 * 3_600_000;
  const payload = `confirmacao_agendamento.${id}.${exp}`;
  const ass = createHmac('sha256', chaveLocal()).update(payload, 'utf8').digest('hex');
  return Buffer.from(`${payload}.${ass}`, 'utf8').toString('base64url');
}

const passos = [];
let n = 0;
async function medir(nome, method, path, body, { anon = false, extras = [], semCorpo = false } = {}) {
  n++;
  const tag = `BA${n}`;
  const headers = anon ? {} : { cookie };
  if (method !== 'GET') {
    headers['content-type'] = 'application/json';
    headers.origin = BASE;
    headers['idempotency-key'] = randomUUID();
  }
  marca(`${tag}-INI`);
  const t0 = performance.now();
  const r = await fetch(BASE + path, { method, headers, body: body && !semCorpo ? JSON.stringify(body) : undefined, redirect: 'manual' });
  const ttfb = performance.now() - t0;
  const texto = await r.text();
  const total = performance.now() - t0;
  marca(`${tag}-RESP`);
  await sleep(2500);
  marca(`${tag}-FIM`);
  let json = null;
  try {
    json = JSON.parse(texto);
  } catch {}
  passos.push({ tag, nome, method, path: path.replace(/\/confirmar\/[^/?]+/, '/confirmar/<token>').replace(/confirm\/[^/?]+/, 'confirm/<token>'), status: r.status, ttfbMs: Math.round(ttfb), totalMs: Math.round(total), extras, erro: r.status >= 400 ? texto.slice(0, 300) : undefined });
  console.log(`${nome}: ${r.status} ${Math.round(total)} ms`);
  return json;
}

function ligarLog() {
  psql('alter system set log_min_duration_statement = 0', 'supabase_admin');
  psql('select pg_reload_conf()', 'supabase_admin');
}
function desligarLog() {
  psql('alter system reset log_min_duration_statement', 'supabase_admin');
  psql('select pg_reload_conf()', 'supabase_admin');
}

const entradas = parse(logsDesde(desde));
const saida = [];
for (const p of passos) {
  const ids = { ...IDS, extras: p.extras };
  const crit = idas(execucoesEntre(entradas, `${p.tag}-INI`, `${p.tag}-RESP`), ids);
  const depois = idas(execucoesEntre(entradas, `${p.tag}-RESP`, `${p.tag}-FIM`), ids);
  const nossos = crit.rest.filter((r) => r.dono === 'nos');
  const todas = [...nossos, ...crit.auth.map((a) => ({ ...a, consultas: ['auth.getUser'] }))];
  const on = ondas(todas);
  saida.push({
    nome: p.nome,
    method: p.method,
    path: p.path,
    status: p.status,
    ttfbMs: p.ttfbMs,
    totalMs: p.totalMs,
    erro: p.erro,
    idasRest: nossos.length,
    idasAuth: crit.auth.length,
    idasTotal: nossos.length + crit.auth.length,
    ondasMedidas: on.length,
    porOnda: on.map((o) => o.itens.map((i) => (i.consultas || []).join(',') + (i.papel && i.papel !== 'authenticated' ? `[${i.papel}]` : ''))),
    restIndefinidas: crit.rest.filter((r) => r.dono === 'indefinido').map((r) => `${r.papel}:${r.consultas.join(',')}`),

  });
